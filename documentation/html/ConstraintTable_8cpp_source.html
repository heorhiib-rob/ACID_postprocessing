26void ConstraintTable::add_constraint(const TimePoint& timepoint, int agent_num)
27{
28 // check the location is valid
29 assertm(instance.get_map_data().is_in(timepoint.location), "Invalid location.");
30 assertm(agent_num >= 0 && agent_num < instance.get_num_of_agents(), "Invalid agent number.");
31 int free_location = instance.location_to_free_location(timepoint.location);
32 std::list<std::pair<TimeInterval, int>>& cur_TI_list = constraints[free_location];
33
34 bool inserted = false;
35 // find where to insert the interval
36 for (auto it = cur_TI_list.begin(); it != cur_TI_list.end(); it++)
37 {
38 // check that the intervals do not have any overlap
39 // Special exception for SIPPS: overlap may exist if
40
41 assertm(!overlap((*it).first, timepoint.interval), "Cannot add overlapping constraints without enabling SIPPS (soft collisions).");
42
43 // insert before the first interval that starts later
44 if (it->first.t_min > timepoint.interval.t_min)
45 {
46 std::pair<TimeInterval, int> new_cons = {timepoint.interval, agent_num};
47 cur_TI_list.insert(it, new_cons);
48 inserted = true;
49 break;
50 }
51 }
52
53 // if no interval starts later, insert to the back of the list
54 if (!inserted)
55 {
56 std::pair<TimeInterval, int> new_cons = {timepoint.interval, agent_num};
57 cur_TI_list.emplace_back(new_cons);
58 }
59
60 // add to the set of visiting agents
61 // if the agent is not yet in the set, add it with count 1
62 if (agents_counts[free_location].find(agent_num) == agents_counts[free_location].end())
63 {
64 agents_counts[free_location][agent_num] = 1;
65 }
66 else
67 {
68 // else increase the count
69 agents_counts[free_location][agent_num] += 1;
70 }
71}

73void ConstraintTable::remove_constraint(const TimePoint& timepoint, int agent_num)
74{
75 // check the location is valid
76 assertm(instance.get_map_data().is_in(timepoint.location), "Invalid location.");
77 int free_location = instance.location_to_free_location(timepoint.location);
78
79
80 std::list<std::pair<TimeInterval, int>>& cur_TI_list = constraints[free_location];
81
82 // check that the list is not empty
83 assertm(!cur_TI_list.empty(), "Trying to remove interval from an empty list.");
84
85 assertm(agents_counts[free_location].find(agent_num) != agents_counts[free_location].end(),
86 "Removing agent that is not in the agents counts");
87 agents_counts[free_location][agent_num] -= 1;
88 // if the count drops to 0, erase the agent number from keys
89 if (agents_counts[free_location][agent_num] == 0)
90 {
91 agents_counts[free_location].erase(agent_num);
92 }
93
94 // remove the interval from the list
95 for (auto it = cur_TI_list.begin(); it != cur_TI_list.end(); it++)
96 {
97 // diferent handling for SIPPS (soft collisions)
98 if ((*it).first == timepoint.interval )
99 {
100 cur_TI_list.erase(it);
101 return;
102 }
103 }
104 // if the interval was not found, throw an error
105 throw std::runtime_error("Trying to remove non-existing interval from the constraint table.");
106}

109void ConstraintTable::add_constraints(const TimePointPath& path, int agent_num)
110{
111 for (int i = 0; i < (int)path.size(); i++)
112 {
113 // add the point constraint
114 // std::cout << "Adding: " << path[i].location << ", " << path[i].interval << std::endl;
115 add_constraint(path[i], agent_num);
116
117 // add the edge constraint
118 if (i > 0) // skip the first point as there is no edge
119 {
120 // the time when the edge is used is the start time of the later interval
121 edge_constraint_table.add(path[i - 1].location, path[i].location, path[i].interval.t_min, agent_num);
122 }
123 }
124}

145auto ConstraintTable::get_blocking_agent(int from, int to, int time) const -> std::pair<int, int>
146{
147 // check the location is valid
148 assertm(instance.get_map_data().is_in(from) && instance.get_map_data().is_in(to), "Invalid location.");
149
150 // check the time is valid
151 assertm(time >= 0, "Invalid time.");
152
153 int to_free = instance.location_to_free_location(to);
154
155 const std::list<std::pair<TimeInterval, int>>& cur_TI_list = constraints[to_free];
156
157 // if there are no constraints, return -1 (no agent blocking the square)
158 if (cur_TI_list.empty())
159 {
160 return std::make_pair(-1, -1);
161 };
162
163 // find the constraint
164 int vertex_constraint = -1;
165 for (const auto& it : cur_TI_list)
166 {
167 const TimeInterval& constraint_tp = it.first;
168 if (constraint_tp.t_min <= time && constraint_tp.t_max >= time)
169 {
170 // return the agent that created the constraint
171 vertex_constraint = it.second;
172 break;
173 }
174 // skip constraints that are too late
175 if (constraint_tp.t_min > time)
176 {
177 break;
178 }
179 }
180
181 // find edge constraint
182 int edge_constraint = -1;
183 if (from != to)
184 {
185 // we want to look for opposite edge, because there can not be any edge ending in to location without location constraint
186 edge_constraint = edge_constraint_table.get(to, from, time);
187 }
188 return std::make_pair(vertex_constraint, edge_constraint);
189}

191auto ConstraintTable::get_blocking_agents(int location, int time_min) const -> std::vector<int>
192{
193 // check the location is valid
194 assertm(instance.get_map_data().is_in(location), "Invalid location.");
195
196 // check the time is valid
197 assertm(time_min >= 0, "Invalid time.");
198
199 int to_free = instance.location_to_free_location(location);
200
201 const std::list<std::pair<TimeInterval, int>>& cur_TI_list = constraints[to_free];
202
203 // if there are no constraints, return -1 (no agent blocking the square)
204 assertm(!cur_TI_list.empty(), "Trying to get blocking agents from an empty list.");
205
206 std::unordered_set<int> blocking_agents = {};
207 std::vector<int> blocking_agents_vec = {};
208 assertm(cur_TI_list.back().first.t_max == INT_MAX, "Last interval should be infinite.");
209
210 // check how many intervals are present
211 if (cur_TI_list.size() < 2)
212 {
213 return {};
214 }
215
216 // iterate over the constraints from end to beginning (skip the last interval, as agent can not block itself)
217 for (auto it = std::next(cur_TI_list.rbegin()); it != cur_TI_list.rend(); it++)
218 {
219 // std::cout << "Interval: " << it->first << ", agent: " << it->second << std::endl;
220 if (it->first.t_max < time_min)
221 {
222 // there can not be any more blocking agents
223 break;
224 }
225
226 // check whether the agent was not already added
227 const int agent_num = it->second;
228 if (blocking_agents.find(agent_num) == blocking_agents.end())
229 {
230 // add the agent to the set
231 blocking_agents.insert(agent_num);
232 blocking_agents_vec.push_back(agent_num);
233 }
234 }
235 return blocking_agents_vec;
236}
