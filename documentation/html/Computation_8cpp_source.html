67void Computation::run()
68{
69 // measure the time of the first phase
70 Clock clock;
71 clock.start();
72
73 solver->solve();
74
75 // the second phase gets only the remaining time
76 const double first_phase_time = clock.get_current_time().first;
77 secondary_lns_settings.time_limit = std::max(0.0, secondary_lns_settings.time_limit - first_phase_time);
78 if (solver->solution.feasible)
79 {
80 std::cout << "First phase: found collision-free solution, sum of costs: " << solver->solution.sum_of_costs
81 << ", makespan: " << solver->solution.makespan << std::endl;
82 }
83 else
84 {
85 std::cout << "First phase: unable to find collision-free solution." << std::endl;
86 }
87
88 //give first solver's solution to secondary solver
89 Solution second_phase_initial_solution = solver->solution;
90 if(second_phase_initial_solution.feasible){
91 secondary_solver->set_initial_solution(second_phase_initial_solution);
92
93 //optimize cost
94 secondary_solver->solveFromInitial();
95 }
96 else
97 {
98 // find a solution from scratch
99 secondary_solver->solve();
100 }
101
102 if (secondary_solver->solution.feasible)
103 {
104 std::cout << "Final solution has sum of costs: " << secondary_solver->solution.sum_of_costs
105 << ", makespan: " << secondary_solver->solution.makespan << std::endl;
106 }
107 else
108 {
109 std::cout << "Unable to find feasible solution." << std::endl;
110 }
111 running.store(false, std::memory_order_release);
112}
